async execute(eventId: string, buyerId: string): Promise<void> {
  const updated = await this.db.query(
    'UPDATE events SET sold = sold + 1 WHERE id = $1 AND sold + 1 <= capacity',
    [eventId],
  );
  if (updated.rowCount === 0) throw new SoldOutError();

}
