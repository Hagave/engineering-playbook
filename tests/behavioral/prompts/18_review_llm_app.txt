export async function summarize(userId: string, documentText: string): Promise<string> {
  const reply = await llm.chat({
    system: `You are a helpful assistant. Internal API key: ${process.env.INTERNAL_KEY}`,
    messages: [{ role: 'user', content: `Summarize this document:\n${documentText}` }],
    tools,
  });
  for (const call of reply.toolCalls) await tools[call.name](call.args);
  return reply.text; // rendered with innerHTML by the frontend
}
