```ts
export class PauseSubscriptionDto {
  @IsISO8601()
  resumeAt!: string;

// src/application/pause-subscription.use-case.ts (new)
async execute(subscriptionId: string, resumeAt: Date): Promise<void> {
  const subscription = await this.subscriptions.findActive(subscriptionId);
  subscription.pause(resumeAt);
  await this.subscriptions.save(subscription);
}
```
