async execute(userId: string, code: string, orderTotal: number): Promise<number> {
  const coupon = await this.coupons.findByCode(code);
  if (!coupon || coupon.used) throw new CouponUnavailableError();
  coupon.used = true;
  coupon.usedBy = userId;
  await this.coupons.save(coupon);
  return orderTotal - coupon.amount;
}
